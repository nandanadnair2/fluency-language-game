    58→    const text = manualText.trim();

    63→      const res = await fetch("/api/translate", {
    64→        method: "POST",

    67→      });
    68→      const translated = await res.json();

    70→        original: text,

    74→        targetLanguage: translated.targetLanguage || "en",
    75→        confidence: 0.90,
    76→      });

   103→    }

   105→

   108→    async (code: string) => {

   124→        reconnectionAttempts: 5,

   133→        setSocketConnecting(false);

   138→        });
   139→      });

   142→        console.log("[Watchtower] Socket disconnected:", reason);
   143→        setSocketConnected(false);
   144→        setSocketConnecting(false);

   146→          setConnectionError("Server disconnected. Try generating a new room code.");
   147→        }
   148→      });
   149→
   150→      socket.on("connect_error", (err) => {
   151→        console.error("[Watchtower] Connection error:", err.message);
   152→        setSocketConnected(false);
   153→        setSocketConnecting(false);
   154→        setConnectionError("Connection failed. Make sure the WebSocket service is running.");
   155→      });
   156→

   158→        console.log("[Watchtower] Received subtitle:", data.text?.slice(0, 50));
   159→
   160→        if (data.translation && data.translation.directTranslation) {
   161→          setSubtitle({
   162→            original: data.text,
   163→            directTranslation: data.translation.directTranslation,
   164→            romanized: data.translation.romanized || data.translation.directTranslation,
   165→            sourceLanguage: data.translation.sourceLanguage || "ja",
   166→            targetLanguage: data.translation.targetLanguage || "en",
   167→            confidence: 0.90,
   168→          });
   169→        } else {
   170→          setIsTranslating(true);
   171→          try {
   172→            const res = await fetch("/api/translate", {
   173→              method: "POST",
   174→              headers: { "Content-Type": "application/json" },
   175→              body: JSON.stringify({ text: data.text }),
   176→            });
   177→            const translated = await res.json();
   178→            setSubtitle({
   179→              original: data.text,
   180→              directTranslation: translated.directTranslation || data.text,
   181→              romanized: translated.romanized || data.text,
   182→              sourceLanguage: translated.sourceLanguage || "ja",
   183→              targetLanguage: translated.targetLanguage || "en",
   184→              confidence: 0.90,
   185→            });
   186→          } catch {
   187→            setSubtitle({
   188→              original: data.text,
   189→              directTranslation: data.text,
   190→              romanized: data.text,
   191→              sourceLanguage: "auto",
   192→              targetLanguage: "en",
   193→              confidence: 0.5,
   194→            });
   195→          } finally {
   196→            setIsTranslating(false);
   197→          }
   198→        }
   199→        setIsExpanded(true);
   200→      });
   201→
   202→      socket.on("member-joined", (data: any) => {
   203→        console.log("[Watchtower] Member joined:", data);
   204→      });
   205→
   206→      socket.on("member-left", (data: any) => {
   207→        console.log("[Watchtower] Member left:", data);
   208→      });
   209→    },
   210→    []
   211→  );
   212→

   215→    return () => {
   216→      if (socketRef.current) {
   217→        socketRef.current.disconnect();
   218→        socketRef.current = null;
   219→      }
   220→    };
   221→  }, []);
   222→

   224→    try {
   225→      const res = await fetch("/api/watchtower", {
   226→        method: "POST",
   227→        headers: { "Content-Type": "application/json" },
   228→        body: JSON.stringify({ action: "create-room" }),
   229→      });

   231→      if (data.code) {
   232→        setRoomCode(data.code);
   233→        setIsRoomActive(true);
   234→        connectSocket(data.code);

   236→      }
   237→    } catch {
   238→      // Fallback
   239→    }
   240→
   241→    const code = String(Math.floor(1000 + Math.random() * 9000));
   242→    setRoomCode(code);
   243→    setIsRoomActive(true);
   244→    connectSocket(code);
   245→  }, [connectSocket]);
   246→
   247→  const disconnectRoom = useCallback(() => {
   248→    if (socketRef.current) {
   249→      socketRef.current.disconnect();
   250→      socketRef.current = null;
   251→    }
   252→    if (intervalRef.current) {
   253→      clearInterval(intervalRef.current);
   254→      intervalRef.current = null;
   255→    }
   256→    setIsRoomActive(false);
   257→    setSocketConnected(false);
   258→    setSocketConnecting(false);
   259→    setConnectionError(null);

   261→    setCurrentSubIndex(0);
   262→    setIsPlaying(false);

   265→

   308→      if (intervalRef.current) {
   309→        clearInterval(intervalRef.current);
   310→        intervalRef.current = null;
   311→      }
   312→    }
   313→
   314→    return () => {
   315→      if (intervalRef.current) {
   316→        clearInterval(intervalRef.current);
   317→      }
   318→    };
   319→  }, [isPlaying, subtitles]);
   320→

   324→      const sub = subtitles[currentSubIndex];

   326→
   327→      fetch("/api/translate", {
   328→        method: "POST",
   329→        headers: { "Content-Type": "application/json" },
   330→        body: JSON.stringify({ text: sub.text }),
   331→      })
   332→        .then((r) => r.json())
   333→        .then((translated) => {
   334→          setSubtitle({
   335→            original: sub.text,
   336→            directTranslation: translated.directTranslation || `[${sub.text}]`,
   337→            romanized: translated.romanized || `[${sub.text}]`,
   338→            sourceLanguage: translated.sourceLanguage || "auto",
   339→            targetLanguage: translated.targetLanguage || "en",
   340→            confidence: 0.90,
   341→          });
   342→        })
   343→        .catch(() => {
   344→          setSubtitle({
   345→            original: sub.text,
   346→            directTranslation: `[${sub.text}]`,
   347→            romanized: `[${sub.text}]`,
   348→            sourceLanguage: "auto",
   349→            targetLanguage: "en",
   350→            confidence: 0.5,
   351→          });
   352→        })
   353→        .finally(() => setIsTranslating(false));
   354→    }
   355→  }, [currentSubIndex, subtitles, roomCode]);
   356→

   358→    (word: TranslationResponse) => {
   359→      import("@/lib/db-vocabulary").then(({ saveWord }) => {
   360→        saveWord({
   361→          original: word.original,
   362→          directTranslation: word.directTranslation,
   363→          romanized: word.romanized,
   364→          sourceLanguage: word.sourceLanguage,
   365→          targetLanguage: word.targetLanguage,
   366→          savedAt: new Date(),
   367→          xpEarned: 3,
   368→          reviewCount: 1,
   369→        });
   370→      });
   371→      addXP(3);
   372→      scanWord();
   373→    },
   374→    [addXP, scanWord]
   375→  );
   376→
   377→  const handleDismiss = useCallback(() => {
   378→    setSubtitle(null);
   379→    setIsExpanded(false);
   380→  }, []);
   381→
   382→  return (
   383→    <div className="flex flex-col gap-4 h-full overflow-y-auto custom-scrollbar pb-4">
   384→      {/* Header card */}
   385→      <div className="p-6 rounded-3xl bg-card shadow-xl border border-border/50">
   386→        <div className="flex items-center gap-3 mb-2">
   387→          <div className="w-12 h-12 rounded-2xl bg-coral/10 flex items-center justify-center">
   388→            <Television size={24} weight="duotone" className="text-coral" />
   389→          </div>
   390→          <div>
   391→            <h2 className="font-serif text-lg font-bold text-charcoal">
   392→              Watchtower Mode
   393→            </h2>
   394→            <p className="text-xs text-muted-foreground">
   395→              Translate Japanese from shows, subtitles & text
   396→            </p>
   397→          </div>
   398→        </div>
   399→        <p className="text-xs text-muted-foreground leading-relaxed">
   400→          Type or paste Japanese text, upload subtitles, or sync with our Chrome Extension
   401→          for live translation while watching shows!
   402→        </p>
   403→      </div>
   404→
   405→      {/* ─── Quick Translate (Manual Input) ─── */}
   406→      <motion.div
   407→        className="p-6 rounded-3xl bg-gradient-to-br from-coral/5 to-butter/10 shadow-xl border border-coral/20"
   408→        initial={{ opacity: 0, y: 20 }}
   409→        animate={{ opacity: 1, y: 0 }}
   410→        transition={{ delay: 0.05 }}
   411→      >
   412→        <div className="flex items-center gap-2 mb-4">
   413→          <div className="w-8 h-8 rounded-xl bg-coral/15 flex items-center justify-center">
   414→            <Keyboard size={18} weight="fill" className="text-coral" />
   415→          </div>
   416→          <div>
   417→            <h3 className="font-serif text-base font-bold text-charcoal">
   418→              Quick Translate
   419→            </h3>
   420→            <p className="text-[10px] text-muted-foreground">
   421→              Type or paste any Japanese text
   422→            </p>
   423→          </div>
   424→        </div>
   425→
   426→        <div className="space-y-3">
   427→          <textarea
   428→            value={manualText}
   429→            onChange={(e) => setManualText(e.target.value)}
   430→            placeholder="Type or paste Japanese text here... e.g. こんにちは、世界！"
   431→            className="w-full p-4 rounded-2xl bg-card border-2 border-border/50 text-base text-charcoal placeholder:text-muted-foreground/50 focus:outline-none focus:border-coral/50 resize-none transition-colors min-h-[80px]"
   432→            rows={3}
   433→          />
   434→          <button
   435→            onClick={handleManualTranslate}
   436→            disabled={!manualText.trim() || isTranslating}
   437→            className="w-full py-3 rounded-2xl bg-coral text-white font-medium shadow-lg hover:shadow-xl transition-all active:scale-95 flex items-center justify-center gap-2 disabled:opacity-40 disabled:cursor-not-allowed"

   439→            {isTranslating ? (
   440→              <SpinnerGap size={18} className="animate-spin" />

   445→          </button>
   446→        </div>
   447→
   448→        {/* Quick examples */}
   449→        <div className="mt-3 flex flex-wrap gap-1.5">
   450→          {["こんにちは", "ありがとう", "いただきます", "大丈夫です"].map((phrase) => (
   451→            <button
   452→              key={phrase}
   453→              onClick={() => setManualText(phrase)}
   454→              className="px-3 py-1 rounded-full bg-card/80 text-xs text-muted-foreground hover:text-coral hover:bg-card transition-all active:scale-95"

   456→              {phrase}

   461→
   462→      {/* ─── SRT Upload ─── */}

   467→          animate={{ opacity: 1, y: 0 }}
   468→          transition={{ delay: 0.1 }}
   469→        >
   470→          <div className="flex items-center gap-2 mb-3">
   471→            <div className="w-8 h-8 rounded-xl bg-sage/15 flex items-center justify-center">
   472→              <FileText size={18} weight="fill" className="text-sage" />
   473→            </div>
   474→            <div>
   475→              <h3 className="font-serif text-base font-bold text-charcoal">
   476→                Subtitle File
   477→              </h3>
   478→              <p className="text-[10px] text-muted-foreground">
   479→                Upload an .SRT file to translate line-by-line
   480→              </p>
   481→            </div>
   482→          </div>
   483→
   484→          <label className="w-full py-3.5 rounded-2xl bg-sage/10 hover:bg-sage/20 cursor-pointer transition-all active:scale-[0.98] flex items-center justify-center gap-2 border border-sage/20">
   485→            <Upload size={18} weight="duotone" className="text-sage" />
   486→            <span className="text-sm font-medium text-sage">
   487→              {isUploading ? "Parsing SRT..." : "Upload .SRT File"}

   505→      >
   506→        {/* Section header — always visible, clickable to expand */}

   526→                For live subtitles from Netflix/YouTube
   527→              </p>

   537→
   538→        {/* Collapsible content */}

   541→            <motion.div
   542→              initial={{ height: 0, opacity: 0 }}

   549→                {/* Setup guide */}

   551→                  <div className="flex items-start gap-3">

   553→                      1
   554→                    </div>
   555→                    <div className="flex-1">
   556→                      <p className="text-sm font-medium text-charcoal">

   567→                        </p>

   572→                  <div className="flex items-start gap-3">

   574→                      2
