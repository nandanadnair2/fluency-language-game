    28→interface ParsedSubtitle {

    32→
    33→export default function WatchtowerTab() {
    34→  const [manualText, setManualText] = useState("");
    35→  const [isTranslating, setIsTranslating] = useState(false);
    36→
    37→  const [roomCode, setRoomCode] = useState<string>("");
    38→  const [isRoomActive, setIsRoomActive] = useState(false);

    44→  const [currentSubIndex, setCurrentSubIndex] = useState(0);

    46→  const [isUploading, setIsUploading] = useState(false);

    48→  const [showExtensionSection, setShowExtensionSection] = useState(false);

    52→
    53→  const addXP = useGameStore((s) => s.addXP);
    54→  const scanWord = useGameStore((s) => s.scanWord);
    55→
    56→  // ─── Manual Text Translation ───
    57→  const handleManualTranslate = useCallback(async () => {
    58→    const text = manualText.trim();

    60→
    61→    setIsTranslating(true);

    63→      const res = await fetch("/api/translate", {
    64→        method: "POST",
    65→        headers: { "Content-Type": "application/json" },
    66→        body: JSON.stringify({ text }),
    67→      });
    68→      const translated = await res.json();
    69→      setSubtitle({
    70→        original: text,
    71→        directTranslation: translated.directTranslation || text,
    72→        romanized: translated.romanized || text,
    73→        sourceLanguage: translated.sourceLanguage || "ja",
    74→        targetLanguage: translated.targetLanguage || "en",
    75→        confidence: 0.90,
    76→      });

    78→    } catch {
    79→      setSubtitle({
    80→        original: text,
    81→        directTranslation: `[Could not translate: ${text}]`,
    82→        romanized: text,
    83→        sourceLanguage: "auto",
    84→        targetLanguage: "en",
    85→        confidence: 0.5,
    86→      });

    90→    }

    92→

   100→    } catch (err) {
   101→      console.error("Failed to load socket.io-client:", err);
   102→      return null;
   103→    }
   104→  }, [socketModule]);
   105→

   108→    async (code: string) => {
   109→      const mod = await loadSocketIO();

   123→        ? `${window.location.protocol}//${window.location.hostname}:3004`
   124→        : "http://localhost:3004";

   126→      const socket = mod.io(socketUrl, {
   127→        path: "/",
   128→        transports: ["websocket", "polling"],
   129→        timeout: 10000,
   130→        reconnection: true,
   131→        reconnectionAttempts: 5,
   132→        reconnectionDelay: 2000,
   133→      });

   135→      socketRef.current = socket;
   136→

   138→        console.log("[Watchtower] Socket connected:", socket.id);
   139→        setSocketConnected(true);
   140→        setSocketConnecting(false);
   141→        setConnectionError(null);
   142→
   143→        socket.emit("join-room", code, (response: any) => {
   144→          console.log("[Watchtower] Joined room:", code, response);
   145→        });
   146→      });
   147→

   149→        console.log("[Watchtower] Socket disconnected:", reason);
   150→        setSocketConnected(false);
   151→        setSocketConnecting(false);
   152→        if (reason === "io server disconnect") {
   153→          setConnectionError("Server disconnected. Try generating a new room code.");
   154→        }
   155→      });
   156→
   157→      socket.on("connect_error", (err) => {
   158→        console.error("[Watchtower] Connection error:", err.message);
   159→        setSocketConnected(false);
   160→        setSocketConnecting(false);
   161→        setConnectionError("Connection failed. Make sure the WebSocket service is running.");
   162→      });
   163→
   164→      socket.on("subtitle", async (data: { code: string; text: string; translation: any }) => {
   165→        console.log("[Watchtower] Received subtitle:", data.text?.slice(0, 50));
   166→
   167→        if (data.translation && data.translation.directTranslation) {
   168→          setSubtitle({
   169→            original: data.text,
   170→            directTranslation: data.translation.directTranslation,
   171→            romanized: data.translation.romanized || data.translation.directTranslation,
   172→            sourceLanguage: data.translation.sourceLanguage || "ja",
   173→            targetLanguage: data.translation.targetLanguage || "en",
   174→            confidence: 0.90,
   175→          });
   176→        } else {
   177→          setIsTranslating(true);
   178→          try {
   179→            const res = await fetch("/api/translate", {
   180→              method: "POST",
   181→              headers: { "Content-Type": "application/json" },
   182→              body: JSON.stringify({ text: data.text }),
   183→            });
   184→            const translated = await res.json();
   185→            setSubtitle({
   186→              original: data.text,
   187→              directTranslation: translated.directTranslation || data.text,
   188→              romanized: translated.romanized || data.text,
   189→              sourceLanguage: translated.sourceLanguage || "ja",
   190→              targetLanguage: translated.targetLanguage || "en",
   191→              confidence: 0.90,
   192→            });
   193→          } catch {
   194→            setSubtitle({
   195→              original: data.text,
   196→              directTranslation: data.text,
   197→              romanized: data.text,
   198→              sourceLanguage: "auto",
   199→              targetLanguage: "en",
   200→              confidence: 0.5,
   201→            });
   202→          } finally {
   203→            setIsTranslating(false);
   204→          }
   205→        }
   206→        setIsExpanded(true);
   207→      });
   208→
   209→      socket.on("member-joined", (data: any) => {
   210→        console.log("[Watchtower] Member joined:", data);
   211→      });
   212→
   213→      socket.on("member-left", (data: any) => {
   214→        console.log("[Watchtower] Member left:", data);
   215→      });
   216→    },
   217→    []
   218→  );
   219→
   220→  // Cleanup socket on unmount
   221→  useEffect(() => {
   222→    return () => {
   223→      if (socketRef.current) {
   224→        socketRef.current.disconnect();
   225→        socketRef.current = null;
   226→      }
   227→    };
   228→  }, []);
   229→

   231→    try {
   232→      const res = await fetch("/api/watchtower", {

   234→        headers: { "Content-Type": "application/json" },
   235→        body: JSON.stringify({ action: "create-room" }),
   236→      });
   237→      const data = await res.json();
   238→      if (data.code) {
   239→        setRoomCode(data.code);
   240→        setIsRoomActive(true);
   241→        connectSocket(data.code);
   242→        return;
   243→      }
   244→    } catch {
   245→      // Fallback
   246→    }
   247→
   248→    const code = String(Math.floor(1000 + Math.random() * 9000));
   249→    setRoomCode(code);
   250→    setIsRoomActive(true);
   251→    connectSocket(code);
   252→  }, [connectSocket]);
   253→
   254→  const disconnectRoom = useCallback(() => {
   255→    if (socketRef.current) {
   256→      socketRef.current.disconnect();
   257→      socketRef.current = null;
   258→    }
   259→    if (intervalRef.current) {
   260→      clearInterval(intervalRef.current);
   261→      intervalRef.current = null;
   262→    }
   263→    setIsRoomActive(false);
   264→    setSocketConnected(false);
   265→    setSocketConnecting(false);
   266→    setConnectionError(null);
   267→    setSubtitles([]);
   268→    setCurrentSubIndex(0);
   269→    setIsPlaying(false);
   270→    setSubtitle(null);
   271→  }, []);
   272→
   273→  const handleSRTUpload = useCallback(
   274→    async (e: React.ChangeEvent<HTMLInputElement>) => {

   285→        });

   288→          setSubtitles(data.subtitles);
   289→          setCurrentSubIndex(0);
   290→          setRoomCode("SRT-LOADED");

   299→    []

   301→
   302→  // Auto-play subtitles

   304→    if (isPlaying && subtitles.length > 0) {

   307→          if (prev + 1 >= subtitles.length) {
   308→            setIsPlaying(false);
   309→            return prev;
   310→          }
   311→          return prev + 1;
   312→        });
   313→      }, 3000);
   314→    } else {
   315→      if (intervalRef.current) {
   316→        clearInterval(intervalRef.current);
   317→        intervalRef.current = null;
   318→      }
   319→    }
   320→
   321→    return () => {
   322→      if (intervalRef.current) {
   323→        clearInterval(intervalRef.current);
   324→      }
   325→    };
   326→  }, [isPlaying, subtitles]);
   327→
   328→  // Translate current SRT subtitle when index changes
   329→  useEffect(() => {
   330→    if (subtitles.length > 0 && currentSubIndex < subtitles.length && roomCode === "SRT-LOADED") {
   331→      const sub = subtitles[currentSubIndex];
   332→      setIsTranslating(true);
   333→
   334→      fetch("/api/translate", {
   335→        method: "POST",
   336→        headers: { "Content-Type": "application/json" },
   337→        body: JSON.stringify({ text: sub.text }),
   338→      })
   339→        .then((r) => r.json())
   340→        .then((translated) => {
   341→          setSubtitle({
   342→            original: sub.text,
   343→            directTranslation: translated.directTranslation || `[${sub.text}]`,
   344→            romanized: translated.romanized || `[${sub.text}]`,
   345→            sourceLanguage: translated.sourceLanguage || "auto",
   346→            targetLanguage: translated.targetLanguage || "en",
   347→            confidence: 0.90,
   348→          });
   349→        })
   350→        .catch(() => {
   351→          setSubtitle({
   352→            original: sub.text,
   353→            directTranslation: `[${sub.text}]`,
   354→            romanized: `[${sub.text}]`,
   355→            sourceLanguage: "auto",
   356→            targetLanguage: "en",
   357→            confidence: 0.5,
   358→          });
   359→        })
   360→        .finally(() => setIsTranslating(false));
   361→    }
   362→  }, [currentSubIndex, subtitles, roomCode]);
   363→
   364→  const handleSave = useCallback(
   365→    (word: TranslationResponse) => {
   366→      import("@/lib/db-vocabulary").then(({ saveWord }) => {
   367→        saveWord({
   368→          original: word.original,
   369→          directTranslation: word.directTranslation,
   370→          romanized: word.romanized,
   371→          sourceLanguage: word.sourceLanguage,
   372→          targetLanguage: word.targetLanguage,
   373→          savedAt: new Date(),
   374→          xpEarned: 3,
   375→          reviewCount: 1,
   376→        });
   377→      });
   378→      addXP(3);
   379→      scanWord();
   380→    },
   381→    [addXP, scanWord]
   382→  );
   383→
   384→  const handleDismiss = useCallback(() => {
   385→    setSubtitle(null);
   386→    setIsExpanded(false);
   387→  }, []);
   388→
   389→  return (
   390→    <div className="flex flex-col gap-4 h-full overflow-y-auto custom-scrollbar pb-4">
   391→      {/* Header card */}
   392→      <div className="p-6 rounded-3xl bg-card shadow-xl border border-border/50">
   393→        <div className="flex items-center gap-3 mb-2">
   394→          <div className="w-12 h-12 rounded-2xl bg-coral/10 flex items-center justify-center">
   395→            <Television size={24} weight="duotone" className="text-coral" />
   396→          </div>
   397→          <div>
   398→            <h2 className="font-serif text-lg font-bold text-charcoal">
   399→              Watchtower Mode
   400→            </h2>
   401→            <p className="text-xs text-muted-foreground">
   402→              Translate Japanese from shows, subtitles & text
   403→            </p>
   404→          </div>
   405→        </div>
   406→        <p className="text-xs text-muted-foreground leading-relaxed">
   407→          Type or paste Japanese text, upload subtitles, or sync with our Chrome Extension
   408→          for live translation while watching shows!
   409→        </p>
   410→      </div>
   411→
   412→      {/* ─── Quick Translate (Manual Input) ─── */}
   413→      <motion.div
   414→        className="p-6 rounded-3xl bg-gradient-to-br from-coral/5 to-butter/10 shadow-xl border border-coral/20"
   415→        initial={{ opacity: 0, y: 20 }}
   416→        animate={{ opacity: 1, y: 0 }}
   417→        transition={{ delay: 0.05 }}
   418→      >
   419→        <div className="flex items-center gap-2 mb-4">
   420→          <div className="w-8 h-8 rounded-xl bg-coral/15 flex items-center justify-center">
   421→            <Keyboard size={18} weight="fill" className="text-coral" />
   422→          </div>
   423→          <div>
   424→            <h3 className="font-serif text-base font-bold text-charcoal">
   425→              Quick Translate
   426→            </h3>
   427→            <p className="text-[10px] text-muted-foreground">
   428→              Type or paste any Japanese text
   429→            </p>
   430→          </div>
   431→        </div>
   432→
   433→        <div className="space-y-3">
   434→          <textarea
   435→            value={manualText}
   436→            onChange={(e) => setManualText(e.target.value)}
   437→            placeholder="Type or paste Japanese text here... e.g. こんにちは、世界！"
   438→            className="w-full p-4 rounded-2xl bg-card border-2 border-border/50 text-base text-charcoal placeholder:text-muted-foreground/50 focus:outline-none focus:border-coral/50 resize-none transition-colors min-h-[80px]"
   439→            rows={3}
   440→          />
   441→          <button
   442→            onClick={handleManualTranslate}
   443→            disabled={!manualText.trim() || isTranslating}
   444→            className="w-full py-3 rounded-2xl bg-coral text-white font-medium shadow-lg hover:shadow-xl transition-all active:scale-95 flex items-center justify-center gap-2 disabled:opacity-40 disabled:cursor-not-allowed"

   446→            {isTranslating ? (
   447→              <SpinnerGap size={18} className="animate-spin" />
   448→            ) : (
   449→              <Translate size={18} weight="bold" />
   450→            )}
   451→            {isTranslating ? "Translating with AI..." : "Translate"}
   452→          </button>
   453→        </div>
   454→
   455→        {/* Quick examples */}
   456→        <div className="mt-3 flex flex-wrap gap-1.5">
   457→          {["こんにちは", "ありがとう", "いただきます", "大丈夫です"].map((phrase) => (
   458→            <button
   459→              key={phrase}
   460→              onClick={() => setManualText(phrase)}
   461→              className="px-3 py-1 rounded-full bg-card/80 text-xs text-muted-foreground hover:text-coral hover:bg-card transition-all active:scale-95"
   462→            >
   463→              {phrase}
   464→            </button>
   465→          ))}
   466→        </div>
   467→      </motion.div>
   468→
   469→      {/* ─── SRT Upload ─── */}
   470→      {!isRoomActive && (
   471→        <motion.div
   472→          className="p-6 rounded-3xl bg-card shadow-xl border border-border/50"
   473→          initial={{ opacity: 0, y: 20 }}
   474→          animate={{ opacity: 1, y: 0 }}
   475→          transition={{ delay: 0.1 }}
   476→        >
   477→          <div className="flex items-center gap-2 mb-3">
   478→            <div className="w-8 h-8 rounded-xl bg-sage/15 flex items-center justify-center">
   479→              <FileText size={18} weight="fill" className="text-sage" />
   480→            </div>
   481→            <div>
   482→              <h3 className="font-serif text-base font-bold text-charcoal">
   483→                Subtitle File
   484→              </h3>
   485→              <p className="text-[10px] text-muted-foreground">
   486→                Upload an .SRT file to translate line-by-line
   487→              </p>
   488→            </div>
   489→          </div>
   490→
   491→          <label className="w-full py-3.5 rounded-2xl bg-sage/10 hover:bg-sage/20 cursor-pointer transition-all active:scale-[0.98] flex items-center justify-center gap-2 border border-sage/20">
   492→            <Upload size={18} weight="duotone" className="text-sage" />
   493→            <span className="text-sm font-medium text-sage">
   494→              {isUploading ? "Parsing SRT..." : "Upload .SRT File"}
   495→            </span>
   496→            <input
   497→              type="file"
   498→              accept=".srt"
   499→              onChange={handleSRTUpload}
   500→              className="hidden"

   505→
   506→      {/* ─── Chrome Extension + Room Code (Collapsible Advanced Section) ─── */}
   507→      <motion.div
   508→        className="rounded-3xl bg-card shadow-xl border border-border/50 overflow-hidden"
   509→        initial={{ opacity: 0, y: 20 }}
   510→        animate={{ opacity: 1, y: 0 }}
   511→        transition={{ delay: 0.15 }}
   512→      >
   513→        {/* Section header — always visible, clickable to expand */}
   514→        <button
   515→          onClick={() => {
   516→            if (showExtensionSection) {
   517→              setShowExtensionSection(false);
   518→            } else {
   519→              setShowExtensionSection(true);
   520→            }
   521→          }}
   522→          className="w-full p-5 flex items-center justify-between text-left hover:bg-secondary/30 transition-colors"
   523→        >
   524→          <div className="flex items-center gap-2">
   525→            <div className="w-8 h-8 rounded-xl bg-butter/20 flex items-center justify-center">
   526→              <PuzzlePiece size={18} weight="fill" className="text-butter" />
   527→            </div>
   528→            <div>
   529→              <h3 className="font-serif text-sm font-bold text-charcoal">
   530→                Chrome Extension Sync
   531→              </h3>
   532→              <p className="text-[10px] text-muted-foreground">
   533→                For live subtitles from Netflix/YouTube
   534→              </p>
   535→            </div>
   536→          </div>
   537→          <motion.div
   538→            animate={{ rotate: showExtensionSection ? 180 : 0 }}
   539→            transition={{ duration: 0.2 }}

   541→            <CaretDown size={18} weight="bold" className="text-muted-foreground" />
   542→          </motion.div>
   543→        </button>
   544→
   545→        {/* Collapsible content */}
   546→        <AnimatePresence>
   547→          {showExtensionSection && (
   548→            <motion.div
   549→              initial={{ height: 0, opacity: 0 }}
   550→              animate={{ height: "auto", opacity: 1 }}
   551→              exit={{ height: 0, opacity: 0 }}

   553→              className="overflow-hidden"
   554→            >
   555→              <div className="px-5 pb-5 space-y-4 border-t border-border/30 pt-4">
   556→                {/* Setup guide */}
   557→                <div className="space-y-3">
   558→                  <div className="flex items-start gap-3">
   559→                    <div className="w-6 h-6 rounded-full bg-coral text-white flex items-center justify-center text-[10px] font-bold shrink-0 mt-0.5">
   560→                      1
   561→                    </div>
   562→                    <div className="flex-1">
   563→                      <p className="text-sm font-medium text-charcoal">
   564→                        Install the Extension
   565→                      </p>
   566→                      <p className="text-xs text-muted-foreground mt-0.5">
   567→                        Load the <span className="font-medium text-coral">Fluency</span> extension from{" "}

   569→                      </p>
   570→                      <div className="mt-1.5 p-2.5 rounded-xl bg-secondary/60">
   571→                        <p className="text-[11px] text-muted-foreground flex items-start gap-1.5">
   572→                          <ArrowRight size={10} weight="bold" className="text-coral shrink-0 mt-0.5" />
   573→                          Chrome → <code className="px-1 py-0.5 rounded bg-card text-[10px] font-mono">chrome://extensions</code> → Developer mode → Load unpacked
   574→                        </p>
